from pyspark.sql import functions as F
from functools import reduce

# Security table name -> type label
security_tables = {
    "VW_DIM_SECURITY_VALUECHAIN":      "VALUECHAIN",
    "VW_DIM_SECURITY_VP2":             "VP2",
    "VW_DIM_SECURITY_MEGAREGION":      "MEGAREGION",
    "VW_DIM_SECURITY_PRODUCTGROUP":    "PRODUCTGROUP",
    "VW_DIM_SECURITY_MEASURES_BLUE":   "BLUE",
    "VW_DIM_SECURITY_MEASURES_GREEN":  "GREEN",
    "VW_DIM_SECURITY_MEASURES_RED":    "RED",
    "VW_DIM_SECURITY_MEASURES_PURPLE": "PURPLE",
    "VW_DIM_SECURITY_MEASURES_ORANGE": "ORANGE",
}

NTID_COL  = "NTID_UPN"   # change if the lakehouse column name is different
VALUE_COL = "VALUE"

dfs = []
for table_name, sec_type in security_tables.items():
    df = spark.read.table(table_name)

    # Stop early if a column name doesn't match
    missing = [c for c in (NTID_COL, VALUE_COL) if c not in df.columns]
    if missing:
        raise ValueError(f"{table_name} is missing {missing}. Columns found: {df.columns}")

    dfs.append(
        df.select(
            F.lower(F.trim(F.col(NTID_COL))).alias("ntid"),
            F.lit(sec_type).alias("security_type"),
            F.trim(F.col(VALUE_COL).cast("string")).alias("allowed_value"),
            F.lit(table_name).alias("source_table"),
        )
    )

# Combine all 9, remove blanks and duplicates
dpm_access = (
    reduce(lambda a, b: a.unionByName(b), dfs)
    .filter(F.col("ntid").isNotNull() & (F.col("ntid") != ""))
    .filter(F.col("allowed_value").isNotNull() & (F.col("allowed_value") != ""))
    .dropDuplicates(["ntid", "security_type", "allowed_value"])
    .withColumn("load_ts", F.current_timestamp())
)

# Save as a Delta table in the lakehouse
(dpm_access.write
    .mode("overwrite")
    .option("overwriteSchema", "true")
    .format("delta")
    .saveAsTable("app_dpm_user_access"))

print("Rows written:", dpm_access.count())
